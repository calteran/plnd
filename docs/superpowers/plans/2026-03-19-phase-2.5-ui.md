# Phase 2.5 UI Design Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the Phase 2.5 visual design — design tokens, responsive navigation, and the meal-anchored daily schedule view — on top of the existing Angular 21 frontend.

**Architecture:** CSS custom properties provide the token layer (light/dark), consumed by SCSS files in new schedule components. The existing `/dashboard` route is replaced with a full schedule view composed from focused sub-components. A mock `ScheduleService` provides data; the backend API is deferred.

**Tech Stack:** Angular 21, signals (`signal`, `computed`, `effect`, `input`, `output`), SCSS, Jasmine/Karma via `just test-ng-headless`

```typescript
import { Component, effect, inject } from '@angular/core';
import { RouterOutlet } from '@angular/router';
import { PreferencesService } from './core/services/preferences.service';

@Component({
  selector: 'app-root',
  imports: [RouterOutlet],
  template: `<router-outlet />`
})
export class AppComponent {
  private readonly preferencesService = inject(PreferencesService);

  constructor() {
    effect(() => {
      const prefs = this.preferencesService.preferences();
      const theme = prefs?.theme ?? 'system';
      const prefersDark = window.matchMedia('(prefers-color-scheme: dark)').matches;
      const useDark = theme === 'dark' || (theme === 'system' && prefersDark);
      document.documentElement.setAttribute('data-theme', useDark ? 'dark' : 'light');
    });
  }
}
```

```typescript
  duringMealTasks: ScheduleTask[];
}

export type ScheduleItem = ScheduleTask | MealAnchor;

export function isMealAnchor(item: ScheduleItem): item is MealAnchor {
  return 'emoji' in item;
}
```

```typescript
// plnd_client/src/app/core/services/schedule.service.ts
import { Injectable, signal } from '@angular/core';
import { ScheduleItem, MealAnchor, ScheduleTask, ItemStatus, isMealAnchor } from '../models/schedule';

@Injectable({ providedIn: 'root' })
export class ScheduleService {
  private readonly _items   = signal<ScheduleItem[]>([]);
  private readonly _loading = signal(false);
  private readonly _error   = signal<string | null>(null);
  private readonly _date    = signal(todayIso());

  readonly scheduleItems = this._items.asReadonly();
  readonly isLoading     = this._loading.asReadonly();
  readonly loadError     = this._error.asReadonly();
  readonly date          = this._date.asReadonly();

  loadDay(date: string): void {
    this._date.set(date);
    this._loading.set(true);
    this._error.set(null);

    // TODO: replace with HttpClient GET /schedule?date=<date>
    setTimeout(() => {
      this._items.set(mockItems(date));
      this._loading.set(false);
    }, 300);
  }

  toggleItem(id: string): void {
    this._items.update(items => items.map(item => {
      if (isMealAnchor(item)) {
        return {
          ...item,
          duringMealTasks: item.duringMealTasks.map(t =>
            t.id === id ? { ...t, status: flip(t.status) } : t
          )
        };
      }
      return item.id === id ? { ...item, status: flip(item.status) } : item;
    }));
  }
}

function todayIso(): string {
  return new Date().toISOString().split('T')[0];
}

function flip(status: ItemStatus): ItemStatus {
  return status === 'done' ? 'pending' : 'done';
}

function mockItems(date: string): ScheduleItem[] {
  const isToday = date === todayIso();
  const done = (s: ItemStatus) => isToday ? 'done' : s;

  return [
    { id: 't1', type: 'task', label: '🧘 5-min stretch', status: done('pending') },
    {
      id: 'm1', name: 'Breakfast', emoji: '🥑',

        { id: 't-dm2', type: 'task', label: '📖 Read 10 pages',    status: 'pending' }

    { id: 't5', type: 'task',        label: '📞 Call pharmacy',            status: 'pending' },
    { id: 'm2', name: 'Lunch',  emoji: '🥗', status: 'unplanned', duringMealTasks: [] },
```

```typescript
// day-header.ts
import { Component, computed, input, output } from '@angular/core';

@Component({
  selector: 'app-day-header',
  standalone: true,
  template: `
    <div class="day-header">
      <button class="nav-btn" (click)="prev()" aria-label="Previous day">‹</button>
      <div class="date-center" role="button" tabindex="0" (click)="onDateClick()" (keydown.enter)="onDateClick()">
        <span class="date-text">{{ formattedDate() }}</span>
        @if (isToday()) {
          <span class="today-badge">Today</span>
        }
      </div>
      <button class="nav-btn" (click)="next()" aria-label="Next day">›</button>
    </div>
  `,
  styleUrl: './day-header.scss'
})
export class DayHeaderComponent {
  readonly date       = input.required<string>();        // ISO YYYY-MM-DD
  readonly dateChange = output<string>();

  readonly formattedDate = computed(() => {
    const d = new Date(this.date() + 'T00:00:00');
    return d.toLocaleDateString('en-US', { weekday: 'long', month: 'short', day: 'numeric' });
  });

  readonly isToday = computed(() =>
    this.date() === new Date().toISOString().split('T')[0]
  );

  prev(): void { this.dateChange.emit(this.offset(-1)); }
  next(): void { this.dateChange.emit(this.offset(1)); }

  onDateClick(): void {
    // TODO: open date picker
  }

  private offset(days: number): string {
    const d = new Date(this.date() + 'T00:00:00');
    d.setDate(d.getDate() + days);
    return d.toISOString().split('T')[0];
  }
}
```

```typescript
// add-entry.ts
import { Component, output, signal, computed } from '@angular/core';
import { ReactiveFormsModule, FormControl } from '@angular/forms';

type ParsedType = 'task' | 'appointment' | 'exercise';

const EXERCISE_KEYWORDS = /\b(run|walk|jog|cycle|swim|workout|gym|stretch|yoga|hike|bike)\b/i;
const TIME_PATTERN       = /\b\d{1,2}(:\d{2})?\s*(am|pm)\b/i;

@Component({
  selector: 'app-add-entry',
  standalone: true,
  imports: [ReactiveFormsModule],
  template: `
    @if (!isOpen()) {
      <div class="add-row" (click)="open()" role="button" tabindex="0"
           (keydown.enter)="open()" aria-label="Add something to your day">
        <div class="gutter"></div>
        <div class="add-checkbox" aria-hidden="true">+</div>
        <span class="add-label">Add something…</span>
      </div>
    } @else {
      <div class="entry-panel">
        <input
          class="entry-input"
          [value]="_text()"
          (input)="_text.set($any($event.target).value)"
          placeholder="e.g. vitamins with breakfast, walk after lunch, dentist 2pm…"
          (keydown.escape)="close()"
          (keydown.enter)="submit()"
          aria-label="Add something to your day"
          autofocus
        />

        @if (parsedType() === 'appointment') {
          <div class="morph-form morph-form--appointment">
            <input class="morph-field" type="time" [formControl]="timeControl" aria-label="Appointment time" />
            <input class="morph-field" [formControl]="locationControl" placeholder="Location (optional)" aria-label="Location" />
          </div>
        }

        @if (parsedType() === 'exercise') {
          <div class="morph-form morph-form--exercise">
            <input class="morph-field" type="number" [formControl]="durationControl"
                   placeholder="Duration (minutes)" aria-label="Duration in minutes" min="1" />
          </div>
        }

        @if (parseHint()) {
          <p class="parse-hint">{{ parseHint() }}</p>
        }

        <div class="entry-actions">
          <button class="btn-cancel" type="button" (click)="close()">Cancel</button>
          <button class="btn-add" type="button" (click)="submit()" [disabled]="!_text().trim()">Add</button>
        </div>
      </div>
    }
  `,
  styleUrl: './add-entry.scss'
})
export class AddEntryComponent {
  readonly added = output<{ label: string; type: ParsedType; time?: string; location?: string; duration?: number }>();

  readonly isOpen          = signal(false);
  // _text is a signal (not FormControl) so computed() tracks it reactively on every keystroke
  protected readonly _text = signal('');
  readonly timeControl     = new FormControl('');
  readonly locationControl = new FormControl('');
  readonly durationControl = new FormControl<number | null>(null);

  readonly parsedType = computed<ParsedType>(() => {
    const text = this._text();
    if (TIME_PATTERN.test(text)) return 'appointment';
    if (EXERCISE_KEYWORDS.test(text)) return 'exercise';
    return 'task';
  });

  readonly parseHint = computed(() => {
    const text = this._text();
    if (text.length > 3 && this.parsedType() === 'task') return null; // no hint needed
    return null; // hint shown only on NL parse failure — deferred to LLM integration
  });

  open(): void  { this.isOpen.set(true); }
  close(): void { this.isOpen.set(false); this._text.set(''); }

  submit(): void {
    const label = this._text().trim();
    if (!label) return;
    this.added.emit({
      label,
      type: this.parsedType(),
      time:     this.timeControl.value     || undefined,
      location: this.locationControl.value || undefined,
      duration: this.durationControl.value ?? undefined
    });
    this.close();
  }
}
```

```typescript
import { Component, OnInit, inject } from '@angular/core';
import { ScheduleService } from '../../core/services/schedule.service';

    <div class="schedule-view">

      <app-day-header
        [date]="scheduleService.date()"
        (dateChange)="onDateChange($event)"
      />

      @if (scheduleService.isLoading()) {
        <div class="skeleton-list" aria-label="Loading schedule" aria-busy="true">
          <div class="skeleton skeleton--meal"></div>
          <div class="skeleton skeleton--task"></div>
          <div class="skeleton skeleton--task"></div>
        </div>
      } @else if (scheduleService.loadError()) {
        <div class="error-state" role="alert">
          <p>Couldn't load your day.
            <button class="retry-link" (click)="retry()">Tap to retry.</button>
          </p>
        </div>
      } @else if (scheduleService.scheduleItems().length === 0) {
        <div class="empty-state">
          <p>Nothing planned yet. Add something to get started.</p>
        </div>
      } @else {
        <div class="schedule-list">
          @for (item of scheduleService.scheduleItems(); track item.id) {
            @if (isMealAnchor(item)) {
              <app-meal-row
                [meal]="item"
                (toggle)="scheduleService.toggleItem($event)"
                (planMeal)="onPlanMeal($event)"
              />
            } @else {
              <app-task-row
                [item]="item"
                (toggle)="scheduleService.toggleItem($event)"
              />
            }
          }
          <app-add-entry (added)="onItemAdded($event)" />

})
```
