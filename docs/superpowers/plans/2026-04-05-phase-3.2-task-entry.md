# Phase 3.2 — Intent-Driven Task Entry Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Wire up the `/agent/task` endpoint so users can type free-form task descriptions and have the AI infer properties, schedule the task, and surface only genuinely uncertain fields back to the UI.

**Architecture:** The route handler builds a `ToolRegistry` with memory tools + two new schedule tools, then calls the existing `AgentService::run`. The `CreateTaskTool` is terminal (overrides `is_terminal() → true`), so when the LLM calls it the loop stops and its JSON output becomes the `AgentOutput::content` that the handler parses and returns as `TaskEntryResponse`.

**Tech Stack:** Rust/Axum (plnd_api), plnd_agent, plnd_data (sqlx/PostgreSQL), Angular 19+ standalone components with signals.

/// Create a user-defined task from an AI agent interaction.
/// Inserts into `tasks` (type='custom') and `generic_tasks` (category='user_created').
#[tracing::instrument(skip(pool), fields(table = "tasks", op = "create_user_task"))]
pub async fn create_user_task(
    pool: &PgPool,
    user_id: Uuid,
    title: &str,
    energy_level: &str,
    scheduled_at: DateTime<Utc>,
    duration_minutes: i32,
) -> Result<Task, DatabaseError> {
    let mut tx = pool.begin().await.map_err(DatabaseError::from)?;

    let task: Task = sqlx::query_as::<_, Task>(
        "INSERT INTO tasks (user_id, title, task_type, scheduled_start,
                            duration_minutes, energy_required, status)
         VALUES ($1, $2, 'custom', $3, $4, $5, 'planned')
         RETURNING task_id, user_id, title, task_type, scheduled_start,
                   duration_minutes, energy_required, status, is_milepost,
                   created_at, updated_at",
    )
    .bind(user_id)
    .bind(title)
    .bind(scheduled_at)
    .bind(duration_minutes)
    .bind(energy_level)
    .fetch_one(&mut *tx)
    .await
    .map_err(DatabaseError::from)?;

    sqlx::query(
        "INSERT INTO generic_tasks (task_id, category, priority)
         VALUES ($1, 'user_created', 'normal')",
    )
    .bind(task.task_id)
    .execute(&mut *tx)
    .await
    .map_err(DatabaseError::from)?;

    tx.commit().await.map_err(DatabaseError::from)?;

    Ok(task)
}

/// Owned handle for agent-driven task operations.
///
/// Holds [`PgPool`] internally so `plnd_agent` tools do not need a direct
/// sqlx dependency — sqlx is a private implementation detail of `plnd_data`.
#[derive(Clone)]
pub struct TaskStore {
    pool: PgPool,
}

impl TaskStore {
    pub fn new(pool: PgPool) -> Self {
        Self { pool }
    }

    pub async fn create_user_task(
        &self,
        user_id: Uuid,
        title: &str,
        energy_level: &str,
        scheduled_at: DateTime<Utc>,
        duration_minutes: i32,
    ) -> Result<Task, DatabaseError> {
        create_user_task(&self.pool, user_id, title, energy_level, scheduled_at, duration_minutes)
            .await
    }

    /// Find the first free time slot on or after `after` that fits `duration_minutes`
    /// without conflicting with existing schedule_entries.
    ///
    /// Energy level maps to a preferred time window:
    /// - "high"   → 09:00–11:00
    /// - "medium" → 10:00–13:00
    /// - "low"    → 13:00–17:00
    ///
    /// Checks up to 14 days. Falls back to day 15 at the preferred hour if all
    /// checked days have conflicts.
    pub async fn find_free_slot(
        &self,
        user_id: Uuid,
        energy_level: &str,
        duration_minutes: i32,
        after: NaiveDate,
    ) -> Result<(DateTime<Utc>, DateTime<Utc>), DatabaseError> {
        find_free_slot(&self.pool, user_id, energy_level, duration_minutes, after).await
    }
}

```rust
async fn find_free_slot(
    pool: &PgPool,
    user_id: Uuid,
    energy_level: &str,
    duration_minutes: i32,
    after: NaiveDate,
) -> Result<(DateTime<Utc>, DateTime<Utc>), DatabaseError> {
    let pref_hour: u32 = match energy_level {
        "high" => 9,
        "low" => 13,
        _ => 10, // medium default
    };

    for day_offset in 0i64..14 {
        let date = after + chrono::Duration::days(day_offset);
        let slot_start = date
            .and_hms_opt(pref_hour, 0, 0)
            .expect("valid time")
            .and_utc();
        let slot_end = slot_start + chrono::Duration::minutes(duration_minutes as i64);

        let conflict: Option<uuid::Uuid> = sqlx::query_scalar(
            "SELECT entry_id FROM schedule_entries
             WHERE user_id = $1
               AND status NOT IN ('skipped', 'deferred')
               AND scheduled_start < $3
               AND (scheduled_end IS NULL OR scheduled_end > $2)
             LIMIT 1",
        )
        .bind(user_id)
        .bind(slot_start)
        .bind(slot_end)
        .fetch_optional(pool)
        .await
        .map_err(DatabaseError::from)?;

        if conflict.is_none() {
            return Ok((slot_start, slot_end));
        }
    }

    // Fallback: day 15 at preferred hour
    let fallback = after + chrono::Duration::days(14);
    let slot_start = fallback
        .and_hms_opt(pref_hour, 0, 0)
        .expect("valid time")
        .and_utc();
    let slot_end = slot_start + chrono::Duration::minutes(duration_minutes as i64);
    Ok((slot_start, slot_end))
}

// ── Helper ────────────────────────────────────────────────────────────────────

fn db_err(tool: &str, e: plnd_data::db::error::DatabaseError) -> ToolError {
    ToolError::Execution {
        tool: tool.to_string(),
        reason: e.to_string(),
    }
}

fn parse_uuid(input: &serde_json::Value, field: &str, tool: &str) -> Result<Uuid, ToolError> {
    let s = input
        .get(field)
        .and_then(|v| v.as_str())
        .ok_or_else(|| ToolError::InvalidInput {
            tool: tool.to_string(),
            reason: format!("missing required field '{field}'"),
        })?;
    s.parse::<Uuid>().map_err(|_| ToolError::InvalidInput {
        tool: tool.to_string(),
        reason: format!("'{field}' is not a valid UUID"),
    })

// ── FindScheduleSlotTool ──────────────────────────────────────────────────────

pub struct FindScheduleSlotTool {
    store: Arc<TaskStore>,
}

#[async_trait::async_trait]
impl Tool for FindScheduleSlotTool {
    fn name(&self) -> &str {
        "find_schedule_slot"
    }

    fn description(&self) -> &str {
        "Find the first free time slot in the user's schedule that matches the \
         requested energy level and duration. Returns a suggested start/end time. \
         Call this before create_task to pick a concrete schedule time."
    }

    fn schema(&self) -> serde_json::Value {
        serde_json::json!({
            "type": "object",
            "properties": {
                "energy_level": {
                    "type": "string",
                    "enum": ["low", "medium", "high"],
                    "description": "Energy level required for this task."
                },
                "duration_minutes": {
                    "type": "integer",
                    "minimum": 1,
                    "description": "Estimated task duration in minutes."
                },
                "after_date": {
                    "type": "string",
                    "format": "date",
                    "description": "Earliest date to search from (YYYY-MM-DD). Use today's date."
                }
            },
            "required": ["energy_level", "duration_minutes", "after_date"]
        })
    }

    async fn execute(
        &self,
        ctx: &ToolContext,
        input: serde_json::Value,
    ) -> Result<serde_json::Value, ToolError> {
        let energy = input
            .get("energy_level")
            .and_then(|v| v.as_str())
            .ok_or_else(|| ToolError::InvalidInput {
                tool: self.name().to_string(),
                reason: "missing 'energy_level'".to_string(),
            })?;

        let duration = input
            .get("duration_minutes")
            .and_then(|v| v.as_i64())
            .ok_or_else(|| ToolError::InvalidInput {
                tool: self.name().to_string(),
                reason: "missing 'duration_minutes'".to_string(),
            })? as i32;

        let after_str = input
            .get("after_date")
            .and_then(|v| v.as_str())
            .ok_or_else(|| ToolError::InvalidInput {
                tool: self.name().to_string(),
                reason: "missing 'after_date'".to_string(),
            })?;

        let after = NaiveDate::parse_from_str(after_str, "%Y-%m-%d").map_err(|_| {
            ToolError::InvalidInput {
                tool: self.name().to_string(),
                reason: format!("invalid date '{after_str}'; expected YYYY-MM-DD"),
            }
        })?;

        let (start, end) = self
            .store
            .find_free_slot(ctx.user_id, energy, duration, after)
            .await
            .map_err(|e| db_err(self.name(), e))?;

        Ok(serde_json::json!({
            "start": start.to_rfc3339(),
            "end":   end.to_rfc3339(),
            "date":  start.format("%Y-%m-%d").to_string()
        }))
    }
}

                            "type": { "type": "string", "enum": ["show_checkbox", "show_select"] },
                            "field": { "type": "string", "description": "Field name this signal is for." },
                            "label": { "type": "string", "description": "Human-readable label." },
                            "default": { "description": "Suggested default value." },
                            "options": {
                                "type": "array",

                            }

        })
    }

    fn is_terminal(&self) -> bool {
        true
    }

    async fn execute(
        &self,
        ctx: &ToolContext,
        input: serde_json::Value,
    ) -> Result<serde_json::Value, ToolError> {
        let title = input
            .get("title")
            .and_then(|v| v.as_str())
            .ok_or_else(|| ToolError::InvalidInput {
                tool: self.name().to_string(),
                reason: "missing 'title'".to_string(),
            })?;

        let energy = input
            .get("energy_level")
            .and_then(|v| v.as_str())
            .ok_or_else(|| ToolError::InvalidInput {
                tool: self.name().to_string(),
                reason: "missing 'energy_level'".to_string(),
            })?;

        let scheduled_at_str = input
            .get("scheduled_at")
            .and_then(|v| v.as_str())
            .ok_or_else(|| ToolError::InvalidInput {
                tool: self.name().to_string(),
                reason: "missing 'scheduled_at'".to_string(),
            })?;

        let scheduled_at = chrono::DateTime::parse_from_rfc3339(scheduled_at_str)
            .map(|dt| dt.with_timezone(&chrono::Utc))
            .map_err(|_| ToolError::InvalidInput {
                tool: self.name().to_string(),
                reason: format!("invalid 'scheduled_at': {scheduled_at_str}"),
            })?;

        let duration = input
            .get("duration_minutes")
            .and_then(|v| v.as_i64())
            .ok_or_else(|| ToolError::InvalidInput {
                tool: self.name().to_string(),
                reason: "missing 'duration_minutes'".to_string(),
            })? as i32;

        let confirmation = input
            .get("confirmation")
            .and_then(|v| v.as_str())
            .unwrap_or("Task scheduled.");

        let ui_signals = input
            .get("ui_signals")
            .cloned()
            .unwrap_or(serde_json::Value::Array(vec![]));

        let task = self
            .store
            .create_user_task(ctx.user_id, title, energy, scheduled_at, duration)
            .await
            .map_err(|e| db_err(self.name(), e))?;

        Ok(serde_json::json!({
            "actions_taken": [{
                "type": "task_created",
                "task_id": task.task_id,
                "scheduled_at": task.scheduled_start
            }],
            "ui_signals": ui_signals,
            "confirmation": confirmation
        }))
    }
}

#[utoipa::path(
    post,
    path = "/agent/task",
    tag = "Agent",
    security(("bearer_auth" = [])),
    request_body = TaskEntryRequest,
    responses(
        (status = 200, description = "Task created", body = TaskEntryResponse),
        (status = 400, description = "Invalid request body"),
        (status = 401, description = "Unauthorized"),
        (status = 500, description = "Agent or database error"),
    )

#[tracing::instrument(skip(state), fields(user_id = %claims.sub))]
pub async fn post_task(
    State(state): State<Arc<AppState>>,
    Extension(claims): Extension<Claims>,
    Json(body): Json<TaskEntryRequest>,
) -> Result<Json<TaskEntryResponse>, StatusCode> {
    let user_id = Uuid::parse_str(&claims.sub).map_err(|_| StatusCode::UNAUTHORIZED)?;
    let pool = state.db.get_pool().clone();

    // Build tool registry for TaskEntry use case.
    let memory_store = Arc::new(MemoryStore::new(pool.clone()));
    let task_store = Arc::new(TaskStore::new(pool));
    let mut tools = ToolRegistry::new();
    register_memory_tools(&mut tools, memory_store);
    register_schedule_tools(&mut tools, task_store);

    // Prefix today's date so the LLM can use it for scheduling.
    let today = Utc::now().date_naive();
    let input = format!("[Today: {today}]\n{}", body.input);

    let ctx = AgentContext {
        user_id,
        use_case: UseCase::TaskEntry,
        input,
    };

    let output = state
        .agent
        .run(&ctx, tools)
        .await
        .map_err(|e| {
            tracing::error!(error = %e, "agent run failed");
            StatusCode::INTERNAL_SERVER_ERROR
        })?;

    // Parse the terminal tool's JSON output.
    let parsed: serde_json::Value = serde_json::from_str(&output.content).map_err(|e| {
        tracing::error!(error = %e, content = %output.content, "agent output is not valid JSON");
        StatusCode::INTERNAL_SERVER_ERROR
    })?;

    let actions_taken: Vec<ActionTaken> = parsed
        .get("actions_taken")
        .and_then(|v| serde_json::from_value(v.clone()).ok())
        .unwrap_or_default();

    let ui_signals = parsed
        .get("ui_signals")
        .and_then(|v| v.as_array())
        .cloned()
        .unwrap_or_default();

    let confirmation = parsed
        .get("confirmation")
        .and_then(|v| v.as_str())
        .unwrap_or("Task scheduled.")
        .to_string();

    Ok(Json(TaskEntryResponse {
        actions_taken,
        ui_signals,
        confirmation,
    }))
}

import { FormsModule } from '@angular/forms';
import { AgentService } from '../../../core/services/agent.service';
import { TaskEntryResponse, UiSignal } from '../../../core/models/agent';

@Component({

  standalone: true,
  imports: [FormsModule],
  templateUrl: './task-entry.html',
  styleUrl: './task-entry.scss',
})
export class TaskEntryComponent {
  private readonly agent = inject(AgentService);

  readonly inputText    = signal('');
  readonly isLoading    = signal(false);
  readonly response     = signal<TaskEntryResponse | null>(null);
  readonly errorMessage = signal<string | null>(null);

  /** Mutable copy of signal values keyed by field name for two-way binding. */
  signalValues: Record<string, unknown> = {};

  submit(): void {
    const input = this.inputText().trim();
    if (!input) return;

    this.isLoading.set(true);
    this.response.set(null);
    this.errorMessage.set(null);
    this.signalValues = {};

    this.agent.createTask({ input }).subscribe({
      next: (res) => {
        this.response.set(res);
        for (const sig of res.ui_signals) {
          this.signalValues[sig.field] = sig.default ?? null;
        }
        this.isLoading.set(false);
      },
      error: () => {
        this.errorMessage.set('Your task couldn\'t be scheduled. Please try again.');
        this.isLoading.set(false);
      },
    });
  }

  isCheckbox(sig: UiSignal): boolean { return sig.type === 'show_checkbox'; }
  isSelect(sig: UiSignal):   boolean { return sig.type === 'show_select'; }
}
```
